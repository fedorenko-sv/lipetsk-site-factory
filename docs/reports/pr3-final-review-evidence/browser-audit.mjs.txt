 function futureDate(site){
  const moscow=new Date(Date.now()+3*3600000);
  const day=new Date(Date.UTC(moscow.getUTCFullYear(),moscow.getUTCMonth(),moscow.getUTCDate()+2));
  while(!site.booking.workingHours.days.includes(day.getUTCDay()||7))day.setUTCDate(day.getUTCDate()+1);
  return day.toISOString().slice(0,10);
 }

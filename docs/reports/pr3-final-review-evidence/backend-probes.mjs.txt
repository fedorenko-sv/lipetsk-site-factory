import { Pool, Client } from 'pg';
import { spawnSync } from 'node:child_process';
import { readFileSync } from 'node:fs';
import { randomUUID } from 'node:crypto';
import { loadRuntimeConfig } from '../../src/server/config/runtime.ts';
import { assertSafeTestDatabaseUrl } from '../../src/server/config/guards.ts';
import { createLead, checkModeReady } from '../../src/server/leads/create-lead.ts';
import { PgLeadStore } from '../../src/server/db/store.ts';
import { PgOutboxRepo, runOutboxBatch } from '../../src/server/notifications/worker.ts';
import { AdapterRegistry } from '../../src/server/notifications/registry.ts';
import { RecordingAdapter } from '../../src/server/notifications/recording.ts';
import { parse as parsePgConnection } from 'pg-connection-string';
import { existsSync, unlinkSync } from 'node:fs';
import path from 'node:path';

const name = `lsf_test_review_${Date.now()}`;
if (!/^lsf_test_review_[0-9]+$/.test(name)) throw new Error('unsafe fixture name');
const base = new URL('postgresql://127.0.0.1:55543/postgres');
base.username = 'lsf_test'; base.password = 'lsf_test_only';
const admin = new Client({connectionString:base.toString()});
await admin.connect();
let pool;
try {
 await admin.query(`CREATE DATABASE ${name}`);
 base.pathname = '/'+name;
 const dsn = assertSafeTestDatabaseUrl(base.toString());
 const migration = spawnSync(process.execPath, ['--experimental-strip-types', 'scripts/backend/migrate.ts'], {
  env:{...process.env,DATABASE_URL:dsn,TEST_DATABASE_URL:dsn,LSF_MIGRATE_TEST_MODE:'1'},encoding:'utf8'
 });
 if (migration.status !== 0) throw new Error('fixture migration failed');
 pool = new Pool({connectionString:dsn,max:5});
 const auto = JSON.parse(readFileSync('examples/sites/auto.json','utf8'));
 const baseEnv = {
   DATABASE_URL:dsn,ALLOWED_SITE_KEYS:'auto',ALLOWED_ORIGINS:'http://127.0.0.1:3025',
   IDEMPOTENCY_HMAC_KEY:randomUUID().replaceAll('-',''),RATE_LIMIT_HMAC_KEY:randomUUID().replaceAll('-',''),
   OUTBOX_SECRET:randomUUID().replaceAll('-',''), ENABLE_LIVE_LEADS:'false',
   SITE_DESTINATIONS_JSON:JSON.stringify({auto:{destinationKey:'demo-auto-v1',version:1}})
 };
 const runtimeFor = adapterId => loadRuntimeConfig({...baseEnv,
  NOTIFICATION_DESTINATIONS_JSON:JSON.stringify({'demo-auto-v1':{adapterId}})
 });
 const body = { siteKey:'auto',configVersion:auto.contentVersion,kind:'booking',
  serviceId:'tire-change',serviceOptions:{'wheel-size':'R16'},
  preferred:{date:'2026-11-10',time:'10:30',timezone:'Europe/Moscow'},
  customer:{phone:'+7'+'0'.repeat(10)},consent:{accepted:true,version:auto.legal.consentVersion},antiBot:{website:''}
 };
 const clock = {now:()=>new Date('2026-11-05T08:00:00Z')};
 const store = new PgLeadStore(pool);
 const bad = await createLead({runtime:runtimeFor('ghost'),siteLoader:()=>auto,store,clock},
  {idempotencyKey:randomUUID(),clientHmac:'review'},body);
 const badRows = await pool.query('SELECT adapter_id FROM notification_outbox WHERE lead_id=$1',[bad.leadId]);
 console.log(JSON.stringify({case:'unregistered adapter at createLead',outcome:bad.kind,
  savedAdapter:badRows.rows[0]?.adapter_id,expected:'DEMO_NOT_READY; no lead/outbox'}));
 await pool.query('DELETE FROM leads WHERE id=$1',[bad.leadId]);

 const runtime = runtimeFor('recording');
 const created = await createLead({runtime,siteLoader:()=>auto,store,clock},
  {idempotencyKey:randomUUID(),clientHmac:'review'},body);
 await pool.query(`UPDATE notification_outbox SET status='processing',attempts=8,
  lease_until=$2,claim_token='old-claim' WHERE lead_id=$1`,[created.leadId,new Date(Date.now()-60000)]);
 const registry = new AdapterRegistry(); registry.register('recording',()=>new RecordingAdapter(pool));
 const counts = await runOutboxBatch(new PgOutboxRepo(pool),registry,{now:()=>new Date()});
 const stranded = await pool.query('SELECT status,attempts,lease_until<now() AS expired FROM notification_outbox WHERE lead_id=$1',[created.leadId]);
 console.log(JSON.stringify({case:'worker crashes on attempt 8; lease expires',counts,row:stranded.rows[0],
  expected:'recover or move to explicit dead; never permanently processing'}));

 const live = structuredClone(auto); live.mode='live';
 live.legal.approved=true; live.legal.operatorName='Synthetic operator';
 const ready = checkModeReady(live,runtime.destinations.get('auto'),{...runtime,enableLiveLeads:true});
 console.log(JSON.stringify({case:'live with recording only, no ingress or retention',outcome:ready,
  expected:{ok:false,code:'LIVE_NOT_READY'}}));

 const override = new URL(dsn); override.searchParams.set('host','example.invalid');
 override.searchParams.set('database','production_probe');
 let guardPassed = true;
 try{assertSafeTestDatabaseUrl(override.toString());}catch{guardPassed=false;}
 const effective = parsePgConnection(override.toString());
 console.log(JSON.stringify({case:'test DSN query overrides effective PostgreSQL target',guardPassed,
  effectiveHost:effective.host,effectiveDatabase:effective.database,expected:'reject before any SQL; parser only, no connection attempted'}));

 const unsafeName = `review_guard_probe_${Date.now()}`;
 if(!/^review_guard_probe_[0-9]+$/.test(unsafeName))throw new Error('unsafe owned fixture');
 await admin.query(`CREATE DATABASE ${unsafeName}`);
 const actualUrl = new URL(dsn); actualUrl.pathname='/'+unsafeName;
 let actual;
 try{
  const migratedWrongTarget = spawnSync(process.execPath,['--experimental-strip-types','scripts/backend/migrate.ts'],{
   env:{...process.env,DATABASE_URL:actualUrl.toString(),TEST_DATABASE_URL:dsn,LSF_MIGRATE_TEST_MODE:'1'},encoding:'utf8'
  });
  actual=new Client({connectionString:actualUrl.toString()});await actual.connect();
  const tables=await actual.query("SELECT count(*)::int AS n FROM information_schema.tables WHERE table_schema='public'");
  console.log(JSON.stringify({case:'migrator validates TEST_DATABASE_URL but connects DATABASE_URL',exit:migratedWrongTarget.status,
   tablesCreatedInNonTestNamedOwnedFixture:tables.rows[0].n,expected:'reject; zero SQL writes'}));
 }finally{
  if(actual)await actual.end();await admin.query(`DROP DATABASE IF EXISTS ${unsafeName}`);
 }

 const repoRoot = process.cwd();
 const exportArg = path.join(path.dirname(repoRoot),'outside-fixture')+path.sep+'..'+path.sep+path.basename(repoRoot)+path.sep+'work'+path.sep+'review'+path.sep+'export-guard-probe.csv';
 const exportTarget = path.resolve(exportArg);
 if(exportTarget!==path.join(repoRoot,'work','review','export-guard-probe.csv'))throw new Error('unexpected export probe target');
 const exportProbe=spawnSync(process.execPath,['--experimental-strip-types','scripts/backend/export-leads.ts','--out',exportArg],{
  env:{...process.env,DATABASE_URL:dsn},encoding:'utf8'
 });
 const inside=existsSync(exportTarget);
 console.log(JSON.stringify({case:'export path outside/../checkout resolves inside Git',exit:exportProbe.status,fileWrittenInsideCheckout:inside,
  expected:'reject before creating file'}));
 if(inside)unlinkSync(exportTarget);
} finally {
 if(pool) await pool.end();
 await admin.query(`DROP DATABASE IF EXISTS ${name}`);
 await admin.end();
}
